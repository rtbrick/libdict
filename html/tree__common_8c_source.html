45void
46tree_node_rot_left(void* Tree, void* Node)
47{
48 tree_node* const n = Node;
49 tree_node* const nr = n->rlink;
50 ASSERT(nr != NULL);
51 if ((n->rlink = nr->llink) != NULL)
52 n->rlink->parent = n;
53 nr->llink = n;
54
55 tree_node* p = n->parent;
56 n->parent = nr;
57 nr->parent = p;
58 *(p == NULL ? &((tree*)Tree)->root : p->llink == n ? &p->llink : &p->rlink) = nr;
59}

128tree_search_node(void* Tree, const void* key)
129{
130 tree* t = Tree;
131 for (tree_node* node = t->root; node;) {
132 const int cmp = t->cmp_func(key, node->key);
133 if (cmp < 0)
134 node = node->llink;
135 else if (cmp)
136 node = node->rlink;
137 else
138 return node;
139 }
140 return NULL;

251size_t
252tree_traverse(void* Tree, dict_visit_func visit, void* user_data)
253{
254 ASSERT(visit != NULL);
255
256 tree* t = Tree;
257 size_t count = 0;
258 if (t->root) {
259 tree_node* node = tree_node_min(t->root);
260 do {
261 ++count;
262 if (!visit(node->key, node->datum, user_data)) break;
263 node = tree_node_next(node);
264 } while (node);
265 }
266 return count;
267}

269bool
270tree_select(void *Tree, size_t n, const void **key, void **datum)
271{
272 tree* t = Tree;
273 if (n >= t->count) {
274 if (key)
275 *key = NULL;
276 if (datum)
277 *datum = NULL;
278 return false;
279 }
280 tree_node* node;
281 if (n >= t->count / 2) {
282 node = tree_node_max(t->root);
283 n = t->count - 1 - n;
284 while (n--)
285 node = tree_node_prev(node);
286 } else {
287 node = tree_node_min(t->root);
288 while (n--)
289 node = tree_node_next(node);
290 }
291 if (key)
292 *key = node->key;
293 if (datum)
294 *datum = node->datum;
295 return true;
296}

304/* Checks one node of tree_verify_common()'s traversal: its key lies strictly
305 * between the bounds inherited from its ancestors (NULL meaning unbounded), and
306 * its children link back to it. */
307static bool
308node_verify_common(const tree* t, const tree_node* node, const void* lower,
309 const void* upper)
310{
311 /* Global key ordering. Comparing a node only against its immediate parent
312 * would accept an entire subtree placed on the wrong side. */
313 if (lower)
314 VERIFY(t->cmp_func(lower, node->key) < 0);
315 if (upper)
316 VERIFY(t->cmp_func(upper, node->key) > 0);
317 if (node->llink)
318 VERIFY(node->llink->parent == node);
319 if (node->rlink)
320 VERIFY(node->rlink->parent == node);
321 return true;
322}

324/* Verifies the invariants every tree shares: root/parent link consistency,
325 * global key ordering, and that |count| matches the number of nodes reachable.
326 *
327 * This is iterative for the same reason the path-length statistics are: a
328 * degenerate sp_tree or tr_tree is as deep as it has nodes, so a recursive
329 * checker would overflow the stack on the very trees most likely to be
330 * malformed. It may only be used by trees with an untagged parent pointer --
331 * hb_tree and rb_tree pack bits into that field and must verify themselves. */
332bool
333tree_verify_common(const void* Tree)
334{
335 const tree* t = Tree;
336
337 VERIFY(t->root == NULL || t->root->parent == NULL);
338 if (t->root) {
339 VERIFY(t->count > 0);
340 } else {
341 VERIFY(t->count == 0);
342 }
343
344 /* (node, lower bound, upper bound) triples still to visit. A NULL bound
345 * means unbounded on that side. */
346 struct frame { const tree_node* node; const void* lower; const void* upper; };
347 struct frame* stack = NULL;
348 size_t size = 0, capacity = 0, count = 0;
349 bool ok = true, oom = false;
350
351#define PUSH(n, lo, hi) \
352 do { \
353 if (size == capacity) { \
354 const size_t cap = capacity ? capacity * 2 : 32; \
355 struct frame* items = MALLOC(cap * sizeof(*items)); \
356 if (!items) { oom = true; break; } \
357 if (stack) { \
358 memcpy(items, stack, size * sizeof(*items)); \
359 FREE(stack); \
360 } \
361 stack = items; \
362 capacity = cap; \
363 } \
364 stack[size].node = (n); \
365 stack[size].lower = (lo); \
366 stack[size].upper = (hi); \
367 ++size; \
368 } while (0)
369
370 if (t->root)
371 PUSH(t->root, NULL, NULL);
372
373 while (ok && !oom && size) {
374 const struct frame top = stack[--size];
375 const tree_node* const node = top.node;
376 if (!node_verify_common(t, node, top.lower, top.upper)) {
377 ok = false;
378 break;
379 }
380 ++count;
381 if (node->llink)
382 PUSH(node->llink, top.lower, node->key);
383 if (!oom && node->rlink)
384 PUSH(node->rlink, node->key, top.upper);
385 }
386#undef PUSH
387 if (stack)
388 FREE(stack);
389
390 if (oom) {
391 fprintf(stderr, "%s:%d (%s) out of memory\n", __FILE__, __LINE__, __func__);
392 return false;
393 }
394 if (!ok)
395 return false;
396 VERIFY(t->count == count);
397 return true;
398}

400size_t
401tree_clear(void* Tree, dict_delete_func delete_func)
402{
403 tree* t = Tree;
404 const size_t count = t->count;
405
406 tree_node* node = t->root;
407 while (node) {
408 if (node->llink || node->rlink) {
409 node = node->llink ? node->llink : node->rlink;
410 continue;
411 }
412 if (delete_func) delete_func(node->key, node->datum);
413 tree_node* const parent = node->parent;
414 /* Unlink before freeing: comparing a pointer to a freed object is UB. */
415 *(parent ? (parent->llink == node ? &parent->llink : &parent->rlink) : &t->root) = NULL;
416 FREE(node);
417 node = parent;
418 }
419 ASSERT(t->root == NULL);
420 t->count = 0;
421 return count;
422}

432/* The three path-length statistics below share one iterative traversal.
433 *
434 * They must not recurse: a degenerate sp_tree or tr_tree (e.g. a treap given a
435 * constant priority function) is as deep as it has nodes, so recursion would
436 * overflow the stack on a few hundred thousand elements. They must also not
437 * walk parent pointers, because hb_tree and rb_tree pack bits into that field
438 * and only their own PARENT() macro may decode it -- so the traversal keeps an
439 * explicit, growable stack of (node, depth) pairs instead.
440 *
441 * Depths are 1-based, i.e. the root is at depth 1. */
442typedef struct {
443 const tree_node* node;
444 size_t depth;
445} node_depth;
446
447typedef struct {
448 node_depth* items;
449 size_t size;
450 size_t capacity;
451} node_stack;
452
453/* Returns false only if memory could not be obtained. */
454static bool
455node_stack_push(node_stack* stack, const tree_node* node, size_t depth)
456{
457 if (stack->size == stack->capacity) {
458 const size_t capacity = stack->capacity ? stack->capacity * 2 : 32;
459 node_depth* items = MALLOC(capacity * sizeof(*items));
460 if (!items)
461 return false;
462 if (stack->items) {
463 memcpy(items, stack->items, stack->size * sizeof(*items));
464 FREE(stack->items);
465 }
466 stack->items = items;
467 stack->capacity = capacity;
468 }
469 stack->items[stack->size].node = node;
470 stack->items[stack->size].depth = depth;
471 ++stack->size;
472 return true;
473}

475/* Computes, over all nodes of |t|:
476 * |min_path| the shallowest depth at which a child link is absent,
477 * |max_path| the greatest node depth,
478 * |total_path| the sum of all node depths.
479 * Any output pointer may be NULL. Returns false if the traversal ran out of
480 * memory, in which case the outputs are not written. */
481static bool
482tree_path_lengths(const tree* t, size_t* min_path, size_t* max_path,
483 size_t* total_path)
484{
485 size_t min_len = 0, max_len = 0, total_len = 0;
486 node_stack stack = { NULL, 0, 0 };
487 bool ok = true;
488
489 if (t->root && !node_stack_push(&stack, t->root, 1))
490 ok = false;
491
492 while (ok && stack.size) {
493 const node_depth top = stack.items[--stack.size];
494 const tree_node* const node = top.node;
495 const size_t depth = top.depth;
496
497 if (depth > max_len)
498 max_len = depth;
499 total_len += depth;
500 if (!node->llink || !node->rlink) {
501 if (!min_len || depth < min_len)
502 min_len = depth;
503 }
504 if (node->llink && !node_stack_push(&stack, node->llink, depth + 1))
505 ok = false;
506 if (ok && node->rlink && !node_stack_push(&stack, node->rlink, depth + 1))
507 ok = false;
508 }
509 if (stack.items)
510 FREE(stack.items);
511
512 if (!ok)
513 return false;
514 if (min_path)
515 *min_path = min_len;
516 if (max_path)
517 *max_path = max_len;
518 if (total_path)
519 *total_path = total_len;
520 return true;
521}
