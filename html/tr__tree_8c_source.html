28/*
29 * cf. [Aragon and Seidel, 1996], [Knuth 1998]
30 *
31 * A treap is a randomized data structure in which each node of the tree has
32 * an associated key and priority. The priority is chosen at random when the
33 * node is inserted into the tree, unless a dict_prio_func supplies it. Each
34 * node is inserted so that the lexicographic order of the keys is preserved,
35 * and the priority of any node is greater than or equal to the priority of
36 * either of its child nodes; in this way the treap is a combination of a tree
37 * and a max-heap. Note the direction: a dict_prio_func should return a LARGER
38 * value for a key that ought to sit closer to the root. In this
39 * implementation, this is accomplished by first inserting the node according
40 * to the lexicographical order of keys as in a normal binary tree, and then,
41 * if needed, sifting the node upwards using a series of rotations until the
42 * heap property of the tree is restored.
43 */

150dict_insert_result
151tr_tree_insert(tr_tree* tree, void* key)
152{
153 int cmp = 0;
154 tr_node* node = tree->root;
155 tr_node* parent = NULL;
156 while (node) {
157 cmp = tree->cmp_func(key, node->key);
158 if (cmp < 0) {
159 parent = node; node = node->llink;
160 } else if (cmp > 0) {
161 parent = node; node = node->rlink;
162 } else
163 return (dict_insert_result) { &node->datum, false };
164 }
165
166 if (!(node = node_new(key)))
167 return (dict_insert_result) { NULL, false };
168 node->prio = tree->prio_func ? tree->prio_func(key) : dict_rand();
169
170 if (!(node->parent = parent)) {
171 ASSERT(tree->root == NULL);
172 ASSERT(tree->count == 0);
173 tree->root = node;
174 } else {
175 if (cmp < 0)
176 parent->llink = node;
177 else
178 parent->rlink = node;
179
180 unsigned rotations = 0;
181 while (parent->prio < node->prio) {
182 ++rotations;
183 if (parent->llink == node)
184 tree_node_rot_right(tree, parent);
185 else
186 tree_node_rot_left(tree, parent);
187 if (!(parent = node->parent))
188 break;
189 }
190 tree->rotation_count += rotations;
191 }
192 tree->count++;
193 return (dict_insert_result) { &node->datum, true };
194}

196static void
197remove_node(tr_tree* tree, tr_node* node)
198{
199 unsigned rotations = 0;
200 while (node->llink && node->rlink) {
201 ++rotations;
202 if (node->llink->prio > node->rlink->prio)
203 tree_node_rot_right(tree, node);
204 else
205 tree_node_rot_left(tree, node);
206 }
207 tree->rotation_count += rotations;
208
209 tr_node* const out = node->llink ? node->llink : node->rlink;
210 tr_node* const parent = node->parent;
211 if (out)
212 out->parent = parent;
213 *(parent ? (parent->llink == node ? &parent->llink : &parent->rlink) : &tree->root) = out;
214
215 FREE(node);
216 tree->count--;
217}

242static tr_node*
243node_new(void* key)
244{
245 tr_node* node = MALLOC(sizeof(*node));
246 if (node) {
247 node->key = key;
248 node->datum = NULL;
249 node->parent = NULL;
250 node->llink = NULL;
251 node->rlink = NULL;
252 /* Overwritten by the caller, but leaving it uninitialized would make a
253 * future second caller inherit heap garbage as a priority. */
254 node->prio = 0;
255 }
256 return node;
257}

259bool
260tr_tree_verify(const tr_tree* tree)
261{
262 /* Search-tree, parent-link and count invariants. Iterative, because a
263 * treap given a poor priority function degenerates into a chain as deep
264 * as it has nodes. */
265 if (!tree_verify_common(tree))
266 return false;
267
268 /* Max-heap property on priorities. Now that the links have been verified
269 * an in-order walk is safe, and it needs no stack of its own. */
270 const tr_node* node = tree->root;
271 while (node && node->llink)
272 node = node->llink;
273 while (node) {
274 if (node->parent)
275 VERIFY(node->prio <= node->parent->prio);
276 if (node->rlink) {
277 node = node->rlink;
278 while (node->llink)
279 node = node->llink;
280 } else {
281 const tr_node* parent = node->parent;
282 while (parent && parent->rlink == node) {
283 node = parent;
284 parent = parent->parent;
285 }
286 node = parent;
287 }
288 }
289 return true;
290}
